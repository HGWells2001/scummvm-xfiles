cmake_minimum_required(VERSION 3.16)
project(xfiles_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(xfiles-neodb engines/xfiles/neodb.cpp engines/xfiles/asset.cpp engines/xfiles/script.cpp)
add_executable(xfiles-inspect tools/inspect.cpp)
target_link_libraries(xfiles-inspect PRIVATE xfiles-neodb)
enable_testing()
add_executable(neodb-tests tests/neodb.cpp)
target_link_libraries(neodb-tests PRIVATE xfiles-neodb)
add_test(NAME neodb-malformed-and-valid COMMAND neodb-tests)
add_executable(asset-tests tests/asset.cpp)
target_link_libraries(asset-tests PRIVATE xfiles-neodb)
add_test(NAME asset-descriptors COMMAND asset-tests)
add_executable(script-tests tests/script.cpp)
target_link_libraries(script-tests PRIVATE xfiles-neodb)
add_test(NAME action-decoding COMMAND script-tests)
